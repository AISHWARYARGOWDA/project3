@Service
public class PayrollService {

    private final PayrollBatchRepository batchRepository;
    private final AccountBalanceRepository balanceRepository;

    public PayrollService(PayrollBatchRepository batchRepository,
                          AccountBalanceRepository balanceRepository) {
        this.batchRepository = batchRepository;
        this.balanceRepository = balanceRepository;
    }

    public PayrollBatch saveBatch(PayrollBatch batch) {
        // Just save batch (Draft / Submitted), no balance change
        return batchRepository.save(batch);
    }

    @Transactional
    public PayrollBatch approveBatch(Long batchId) {
        PayrollBatch batch = batchRepository.findById(batchId)
                .orElseThrow(() -> new RuntimeException("Batch not found"));

        // change status
        batch.setStatus("Approved");

        // calculate total
        double totalAmount = batch.getPayments().stream()
                .mapToDouble(p -> Double.parseDouble(p.getAmount()))
                .sum();

        // deduct balance
        balanceRepository.decrement(
            batch.getInstruction().getDebitAccount(),
            BigDecimal.valueOf(totalAmount)
        );

        return batchRepository.save(batch);
    }

    public List<PayrollBatch> getAllBatches() {
        return batchRepository.findAll();
    }

    public PayrollBatch getBatchById(Long id) {
        return batchRepository.findById(id).orElse(null);
    }
}
